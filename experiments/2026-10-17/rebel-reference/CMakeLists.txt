cmake_minimum_required(VERSION 3.20)
project(sequence_rebel_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -fPIC")
find_package(Torch REQUIRED)
set(REF /tmp/sequence-rebel-reference/csrc/liars_dice)
add_library(liars_dice_lib ${REF}/liars_dice.cc ${REF}/subgame_solving.cc ${REF}/real_net.cc ${REF}/recursive_solving.cc ${REF}/stats.cc)
target_include_directories(liars_dice_lib PUBLIC ${REF})
target_link_libraries(liars_dice_lib PUBLIC ${TORCH_LIBRARIES})
add_executable(recursive_eval ${REF}/recursive_eval.cc)
target_link_libraries(recursive_eval PRIVATE liars_dice_lib)
